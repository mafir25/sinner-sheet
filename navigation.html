Шпаргалка 
# комментарий
Plus  = button "+" max 5          # значение = число нажатий
Rank  = select "Ранг":            # значение = значение варианта
    ZAYIN = 4 "описание" [коррозия = 0]
    HE    = 6 "описание" [коррозия = 1] if Level >= 12
Level = input number "Уровень" default 1 min 1 max 20
Name  = input "Название"
Out   = output "Итог"             # заполняется присваиванием

var pts = 0                       # помнит значение между нажатиями
on Plus:                          # выполняется при нажатии
    pts += 1

if pts > Rank:
    Out = "**Перерасход!** {pts} / {Rank}"
elif pts == Rank:
    Out = "Ровно"
else:
    Out = "Осталось: {Rank - pts}"
    hide(Plus)

Rank.label  Rank.desc  Rank.index  Rank.коррозия
